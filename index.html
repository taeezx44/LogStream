POST /logs/ingest Ingest single log entry ▾
Request Body:
// POST /api/v1/logs/ingest { "service": "auth-service", "level": "ERROR", "message": "Database connection timeout", "host": "server-a", "meta": { "req_id": "abc123", "latency_ms": 342 } } // 200 Response { "ok": true, "id": "log_1720000001_xyz", "indexed_at": "2025-01-01T00:00:01Z" }
POST /logs/bulk Bulk ingest up to 10,000 logs ▾
// POST /api/v1/logs/bulk { "logs": [ { "service": "api-gw", "level": "INFO", "message": "GET /health 200" }, { "service": "worker", "level": "WARN", "message": "Queue depth 8400" } ] } // 200 Response { "ok": true, "ingested": 2, "failed": 0 }
GET /logs/search Full-text search with filters ▾
// GET /api/v1/logs/search Query params: q = "database timeout" // full-text level = "ERROR,WARN" // comma-sep service = "auth-service" host = "server-a" from = "2025-01-01T00:00:00Z" to = "2025-01-01T01:00:00Z" limit = 100 offset = 0 // 200 Response { "total": 342, "hits": [ { "id": "...", "score": 0.97, "log": {...} } ], "took_ms": 3 }
GET /logs/stream SSE realtime stream ▾
// GET /api/v1/logs/stream?level=ERROR&service=auth // Response: text/event-stream (SSE) data: { "id":"log_xyz", "level":"ERROR", "message":"..." } data: { "id":"log_abc", "level":"WARN", "message":"..." } // JS Client const es = new EventSource('/api/v1/logs/stream') es.onmessage = e => console.log(JSON.parse(e.data))
GET /logs/stats Aggregated stats & metrics ▾
// GET /api/v1/logs/stats?from=&to= { "total": 142800, "by_level": { "ERROR":340, "WARN":1820, "INFO":98000 }, "by_service": { "auth":42000, "api-gw":78000 }, "ingest_rate": 148, "index_size_kb": 20480 }
